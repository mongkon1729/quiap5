// ===== ห้องติว ป.5 : สคริปต์รับคะแนนและข้อสอบ (เวอร์ชัน 4) =====
// 1) เปลี่ยนรหัสด้านล่างเป็นรหัสลับของครูเอง (ใช้เปิดหน้าครู) ห้ามบอกนักเรียน
var TEACHER_KEY = 'เปลี่ยนเป็นรหัสลับของครู';

var SHEET_NAME = 'results';
var HEADERS = ['timestamp', 'student', 'examId', 'examTitle', 'score', 'answered',
  'questionCount', 'completed', 'wrongIds', 'durationSec', 'clientId'];

var EXAM_SHEET_NAME = 'ข้อสอบ';
var EXAM_HEADERS = ['ชุดข้อสอบ', 'วิชา', 'เวลา(นาที)', 'หัวข้อ', 'ตัวชี้วัด', 'Bloom', 'คำถาม',
  'ก', 'ข', 'ค', 'ง', 'คำตอบ', 'คำอธิบาย', 'รูปภาพ',
  'ความยาก', 'ระดับชั้น', 'กลุ่มสาระ', 'รหัสวิชา', 'สาระ', 'มาตรฐาน',
  'หน่วยการเรียนรู้', 'เรื่อง', 'ประเภทการสอบ', 'ความยากของชุด'];

// Adds any missing columns to the right of the header row, so older sheets keep working.
function examHeaders_(sheet, wanted) {
  var lastCol = Math.max(sheet.getLastColumn(), 1);
  var header = sheet.getLastRow() > 0 ? sheet.getRange(1, 1, 1, lastCol).getDisplayValues()[0] : [];
  while (header.length && header[header.length - 1] === '') header.pop();
  var missing = wanted.filter(function (h) { return h && header.indexOf(h) < 0; });
  if (missing.length) {
    sheet.getRange(1, header.length + 1, 1, missing.length).setValues([missing]).setFontWeight('bold');
    header = header.concat(missing);
  }
  if (sheet.getFrozenRows() !== 1) sheet.setFrozenRows(1);
  return header;
}

function getExamSheet_() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName(EXAM_SHEET_NAME) || ss.insertSheet(EXAM_SHEET_NAME);
  examHeaders_(sheet, EXAM_HEADERS);
  return sheet;
}

function readExams_() {
  var sheet = getExamSheet_();
  return sheet.getRange(1, 1, Math.max(sheet.getLastRow(), 1), Math.max(sheet.getLastColumn(), 1)).getDisplayValues();
}

function getSheet_() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName(SHEET_NAME) || ss.insertSheet(SHEET_NAME);
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(HEADERS);
    sheet.setFrozenRows(1);
  }
  return sheet;
}

function clip_(value, max) {
  return String(value == null ? '' : value).slice(0, max);
}

function num_(value) {
  var n = Number(value);
  return isFinite(n) ? n : 0;
}

function json_(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}

// data.records: [{ 'ชุดข้อสอบ': ..., 'คำถาม': ..., ... }] one per question
function importExams_(data) {
  if (data.key !== TEACHER_KEY) return json_({ ok: false, error: 'unauthorized' });
  var records = (data.records || []).slice(0, 500);
  var lock = LockService.getScriptLock();
  lock.waitLock(10000);
  try {
    var sheet = getExamSheet_();
    var wanted = [];
    records.forEach(function (r) {
      Object.keys(r).forEach(function (k) { if (wanted.indexOf(k) < 0) wanted.push(clip_(k, 40)); });
    });
    var header = examHeaders_(sheet, wanted);
    var titleCol = header.indexOf('ชุดข้อสอบ');
    var questionCol = header.indexOf('คำถาม');

    var existing = {};
    var lastTitle = '';
    readExams_().slice(1).forEach(function (row) {
      lastTitle = row[titleCol] || lastTitle;
      existing[lastTitle + '|' + row[questionCol]] = true;
    });

    var rows = [];
    records.forEach(function (r) {
      var title = clip_(r['ชุดข้อสอบ'], 200);
      var question = clip_(r['คำถาม'], 1000);
      if (!title || !question || existing[title + '|' + question]) return;
      existing[title + '|' + question] = true;
      rows.push(header.map(function (h) { return clip_(r[h], 1000); }));
    });

    if (rows.length) {
      sheet.getRange(sheet.getLastRow() + 1, 1, rows.length, header.length)
        .setNumberFormat('@').setValues(rows);
    }
    return json_({ ok: true, added: rows.length });
  } finally {
    lock.releaseLock();
  }
}

function doPost(e) {
  var data = JSON.parse(e.postData.contents);
  if (data.action === 'importExams') return importExams_(data);
  if (!data.student || !data.examId) return json_({ ok: false, error: 'missing fields' });
  var lock = LockService.getScriptLock();
  lock.waitLock(10000);
  try {
    var sheet = getSheet_();
    var clientId = clip_(data.clientId, 40);
    var last = sheet.getLastRow();
    if (clientId && last > 1) {
      var ids = sheet.getRange(2, HEADERS.length, last - 1, 1).getValues();
      for (var i = 0; i < ids.length; i++) {
        if (ids[i][0] === clientId) return json_({ ok: true, duplicate: true });
      }
    }
    sheet.appendRow([
      new Date(data.timestamp || Date.now()),
      clip_(data.student, 40),
      clip_(data.examId, 60),
      clip_(data.examTitle, 120),
      num_(data.score),
      num_(data.answered),
      num_(data.questionCount),
      data.completed === true,
      clip_(data.wrongIds, 500),
      num_(data.durationSec),
      clientId
    ]);
  } finally {
    lock.releaseLock();
  }
  return json_({ ok: true });
}

function doGet(e) {
  if (e.parameter.action === 'exams') {
    return json_({ ok: true, version: 4, values: readExams_() });
  }
  if (!e.parameter.key || e.parameter.key !== TEACHER_KEY) {
    return json_({ ok: false, error: 'unauthorized' });
  }
  var values = getSheet_().getDataRange().getValues();
  var rows = [];
  for (var r = 1; r < values.length; r++) {
    var row = {};
    for (var c = 0; c < HEADERS.length; c++) row[HEADERS[c]] = values[r][c];
    rows.push(row);
  }
  return json_({ ok: true, rows: rows });
}
