// ===== ห้องติว ป.5 : สคริปต์รับคะแนนและข้อสอบ (เวอร์ชัน 6) =====
// 1) เปลี่ยนรหัสด้านล่างเป็นรหัสลับของครูเอง (ใช้เปิดหน้าครู) ห้ามบอกนักเรียน
var TEACHER_KEY = 'เปลี่ยนเป็นรหัสลับของครู';

var SHEET_NAME = 'results';
var HEADERS = ['timestamp', 'student', 'examId', 'examTitle', 'score', 'answered',
  'questionCount', 'completed', 'wrongIds', 'durationSec', 'clientId'];

var EXAM_SHEET_NAME = 'ข้อสอบ';
var EXAM_HEADERS = ['ชุดข้อสอบ', 'วิชา', 'เวลา(นาที)', 'หัวข้อ', 'ตัวชี้วัด', 'Bloom', 'คำถาม',
  'ก', 'ข', 'ค', 'ง', 'คำตอบ', 'คำอธิบาย', 'รูปภาพ',
  'ความยาก', 'ระดับชั้น', 'กลุ่มสาระ', 'รหัสวิชา', 'สาระ', 'มาตรฐาน',
  'หน่วยการเรียนรู้', 'เรื่อง', 'ประเภทการสอบ', 'ความยากของชุด', 'สถานะ'];

// Adds any missing columns to the right of the header row, so older sheets keep working.
function examHeaders_(sheet, wanted) {
  var lastCol = Math.max(sheet.getLastColumn(), 1);
  var header = sheet.getLastRow() > 0 ? sheet.getRange(1, 1, 1, lastCol).getDisplayValues()[0] : [];
  while (header.length && header[header.length - 1] === '') header.pop();
  var missing = wanted.filter(function (h) { return h && header.indexOf(h) < 0; });
  if (missing.length) {
    sheet.getRange(1, header.length + 1, 1, missing.length).setValues([missing]).setFontWeight('bold');
    header = header.concat(missing);
  }
  if (sheet.getFrozenRows() !== 1) sheet.setFrozenRows(1);
  return header;
}

function getExamSheet_() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName(EXAM_SHEET_NAME) || ss.insertSheet(EXAM_SHEET_NAME);
  examHeaders_(sheet, EXAM_HEADERS);
  return sheet;
}

function readExams_() {
  var sheet = getExamSheet_();
  return sheet.getRange(1, 1, Math.max(sheet.getLastRow(), 1), Math.max(sheet.getLastColumn(), 1)).getDisplayValues();
}

function getSheet_() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName(SHEET_NAME) || ss.insertSheet(SHEET_NAME);
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(HEADERS);
    sheet.setFrozenRows(1);
  }
  return sheet;
}

function clip_(value, max) {
  return String(value == null ? '' : value).slice(0, max);
}

function num_(value) {
  var n = Number(value);
  return isFinite(n) ? n : 0;
}

function json_(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}

// data.records: [{ 'ชุดข้อสอบ': ..., 'คำถาม': ..., ... }] one per question
function importExams_(data) {
  if (data.key !== TEACHER_KEY) return json_({ ok: false, error: 'unauthorized' });
  var records = (data.records || []).slice(0, 500);
  var lock = LockService.getScriptLock();
  lock.waitLock(10000);
  try {
    var sheet = getExamSheet_();
    var wanted = [];
    records.forEach(function (r) {
      Object.keys(r).forEach(function (k) { if (wanted.indexOf(k) < 0) wanted.push(clip_(k, 40)); });
    });
    var header = examHeaders_(sheet, wanted);
    var titleCol = header.indexOf('ชุดข้อสอบ');
    var questionCol = header.indexOf('คำถาม');

    var existing = {};
    var lastTitle = '';
    readExams_().slice(1).forEach(function (row) {
      lastTitle = row[titleCol] || lastTitle;
      existing[lastTitle + '|' + row[questionCol]] = true;
    });

    var rows = [];
    records.forEach(function (r) {
      var title = clip_(r['ชุดข้อสอบ'], 200);
      var question = clip_(r['คำถาม'], 1000);
      if (!title || !question || existing[title + '|' + question]) return;
      existing[title + '|' + question] = true;
      rows.push(header.map(function (h) { return clip_(r[h], 1000); }));
    });

    if (rows.length) {
      sheet.getRange(sheet.getLastRow() + 1, 1, rows.length, header.length)
        .setNumberFormat('@').setValues(rows);
    }
    return json_({ ok: true, added: rows.length });
  } finally {
    lock.releaseLock();
  }
}

// Finds the sheet row the teacher is editing and checks it still holds the same question,
// so an edit never lands on the wrong row after someone changed the sheet by hand.
function findQuestionRow_(sheet, header, data) {
  var row = Math.floor(num_(data.row));
  var questionCol = header.indexOf('คำถาม');
  if (row < 2 || row > sheet.getLastRow() || questionCol < 0) return 0;
  var current = sheet.getRange(row, questionCol + 1).getDisplayValue();
  return String(current).trim() === String(data.expect || '').trim() ? row : 0;
}

// data: { row, expect, fields: { 'คำถาม': ..., 'ก': ..., 'รูปภาพ': ... } }
function updateQuestion_(data) {
  if (data.key !== TEACHER_KEY) return json_({ ok: false, error: 'unauthorized' });
  var lock = LockService.getScriptLock();
  lock.waitLock(10000);
  try {
    var sheet = getExamSheet_();
    var fields = data.fields || {};
    var header = examHeaders_(sheet, Object.keys(fields).map(function (k) { return clip_(k, 40); }));
    var row = findQuestionRow_(sheet, header, data);
    if (!row) return json_({ ok: false, error: 'moved' });
    Object.keys(fields).forEach(function (k) {
      var col = header.indexOf(k);
      if (col >= 0) sheet.getRange(row, col + 1).setNumberFormat('@').setValue(clip_(fields[k], 1000));
    });
    return json_({ ok: true });
  } finally {
    lock.releaseLock();
  }
}

// Set-level info usually sits only on a set's first row, so pass it down before deleting that row.
function deleteQuestion_(data) {
  if (data.key !== TEACHER_KEY) return json_({ ok: false, error: 'unauthorized' });
  var lock = LockService.getScriptLock();
  lock.waitLock(10000);
  try {
    var sheet = getExamSheet_();
    var header = examHeaders_(sheet, []);
    var row = findQuestionRow_(sheet, header, data);
    if (!row) return json_({ ok: false, error: 'moved' });
    var titleCol = header.indexOf('ชุดข้อสอบ');
    if (row < sheet.getLastRow() && titleCol >= 0) {
      var both = sheet.getRange(row, 1, 2, header.length).getDisplayValues();
      if (both[0][titleCol] && !both[1][titleCol]) {
        var perQuestion = ['หัวข้อ', 'ตัวชี้วัด', 'Bloom', 'คำถาม', 'ก', 'ข', 'ค', 'ง', 'คำตอบ', 'คำอธิบาย', 'รูปภาพ', 'ความยาก'];
        var next = both[1].map(function (v, i) {
          return v || perQuestion.indexOf(header[i]) >= 0 ? v : both[0][i];
        });
        sheet.getRange(row + 1, 1, 1, header.length).setNumberFormat('@').setValues([next]);
      }
    }
    sheet.deleteRow(row);
    return json_({ ok: true });
  } finally {
    lock.releaseLock();
  }
}

var IMAGE_FOLDER = 'ห้องติว ป.5 - รูปข้อสอบ';

// data: { name, mimeType, base64 } -> saves into a Drive folder anyone with the link can view
function uploadImage_(data) {
  if (data.key !== TEACHER_KEY) return json_({ ok: false, error: 'unauthorized' });
  var mime = String(data.mimeType || '');
  if (['image/png', 'image/jpeg', 'image/gif', 'image/webp'].indexOf(mime) < 0) return json_({ ok: false, error: 'type' });
  var bytes = Utilities.base64Decode(String(data.base64 || ''));
  if (!bytes.length || bytes.length > 5 * 1024 * 1024) return json_({ ok: false, error: 'size' });
  var folders = DriveApp.getFoldersByName(IMAGE_FOLDER);
  var folder = folders.hasNext() ? folders.next() : DriveApp.createFolder(IMAGE_FOLDER);
  var file = folder.createFile(Utilities.newBlob(bytes, mime, clip_(data.name, 100) || 'image'));
  file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
  return json_({ ok: true, url: 'https://drive.google.com/thumbnail?id=' + file.getId() + '&sz=w1200' });
}

// ---------- บัญชีนักเรียน (ครูตั้ง username / รหัสผ่านในแท็บ "นักเรียน") ----------
var STUDENT_SHEET_NAME = 'นักเรียน';
var STUDENT_HEADERS = ['username', 'รหัสผ่าน', 'เลขที่', 'ชื่อ', 'ห้อง', 'token', 'progress', 'อัปเดตล่าสุด'];

function getStudentSheet_() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName(STUDENT_SHEET_NAME) || ss.insertSheet(STUDENT_SHEET_NAME);
  var header = examHeaders_(sheet, STUDENT_HEADERS);
  return { sheet: sheet, header: header };
}

function readStudents_() {
  var st = getStudentSheet_();
  var last = st.sheet.getLastRow();
  var rows = last > 1 ? st.sheet.getRange(2, 1, last - 1, st.header.length).getDisplayValues() : [];
  return rows.map(function (r, i) {
    var o = { row: i + 2 };
    st.header.forEach(function (h, c) { o[h] = r[c]; });
    return o;
  }).filter(function (o) { return o.username; });
}

function normUser_(u) {
  return String(u || '').trim().toLowerCase();
}

function findStudent_(username) {
  var key = normUser_(username);
  var list = readStudents_();
  for (var i = 0; i < list.length; i++) if (normUser_(list[i].username) === key) return list[i];
  return null;
}

function setStudentCell_(row, headerName, value) {
  var st = getStudentSheet_();
  var col = st.header.indexOf(headerName);
  if (col >= 0) st.sheet.getRange(row, col + 1).setNumberFormat('@').setValue(value);
}

function publicStudent_(s) {
  return { username: s.username, name: s['ชื่อ'], number: s['เลขที่'], room: s['ห้อง'] };
}

function parseProgress_(text) {
  try { return text ? JSON.parse(text) : null; } catch (e) { return null; }
}

function studentLogin_(data) {
  var s = findStudent_(data.username);
  if (!s || !s['รหัสผ่าน'] || String(s['รหัสผ่าน']).trim() !== String(data.password || '').trim()) {
    Utilities.sleep(800);
    return json_({ ok: false, error: 'wrong' });
  }
  var token = s.token;
  if (!token) {
    token = Utilities.getUuid();
    setStudentCell_(s.row, 'token', token);
  }
  return json_({ ok: true, student: publicStudent_(s), token: token, progress: parseProgress_(s.progress) });
}

function checkToken_(data) {
  var s = findStudent_(data.username);
  return s && s.token && s.token === data.token ? s : null;
}

function studentProgress_(data) {
  var s = checkToken_(data);
  if (!s) return json_({ ok: false, error: 'token' });
  return json_({ ok: true, student: publicStudent_(s), progress: parseProgress_(s.progress) });
}

function saveProgress_(data) {
  var lock = LockService.getScriptLock();
  lock.waitLock(10000);
  try {
    var s = checkToken_(data);
    if (!s) return json_({ ok: false, error: 'token' });
    var text = JSON.stringify(data.progress || {});
    if (text.length > 45000) return json_({ ok: false, error: 'size' });
    var old = parseProgress_(s.progress);
    if (old && (old.updatedAt || 0) > ((data.progress || {}).updatedAt || 0)) return json_({ ok: true, stale: true });
    setStudentCell_(s.row, 'progress', text);
    setStudentCell_(s.row, 'อัปเดตล่าสุด', new Date().toISOString());
    return json_({ ok: true });
  } finally {
    lock.releaseLock();
  }
}

function listStudents_(data) {
  if (data.key !== TEACHER_KEY) return json_({ ok: false, error: 'unauthorized' });
  return json_({ ok: true, students: readStudents_().map(function (s) {
    var p = publicStudent_(s);
    p.password = s['รหัสผ่าน'];
    p.lastSeen = s['อัปเดตล่าสุด'];
    return p;
  }) });
}

// data.students: [{ username, password, number, name, room }] — adds new ones, updates existing ones.
function saveStudents_(data) {
  if (data.key !== TEACHER_KEY) return json_({ ok: false, error: 'unauthorized' });
  var lock = LockService.getScriptLock();
  lock.waitLock(10000);
  try {
    var st = getStudentSheet_();
    var existing = {};
    readStudents_().forEach(function (s) { existing[normUser_(s.username)] = s; });
    var added = 0, updated = 0;
    (data.students || []).slice(0, 300).forEach(function (n) {
      var user = clip_(String(n.username || '').trim(), 40);
      if (!user) return;
      var old = existing[normUser_(user)];
      if (old) {
        if (n.password && n.password !== old['รหัสผ่าน']) {
          setStudentCell_(old.row, 'รหัสผ่าน', clip_(n.password, 40));
          setStudentCell_(old.row, 'token', '');
        }
        if (n.name != null) setStudentCell_(old.row, 'ชื่อ', clip_(n.name, 60));
        if (n.number != null) setStudentCell_(old.row, 'เลขที่', clip_(n.number, 10));
        if (n.room != null) setStudentCell_(old.row, 'ห้อง', clip_(n.room, 20));
        updated++;
      } else {
        if (!n.password) return;
        var row = st.header.map(function (h) {
          return { 'username': user, 'รหัสผ่าน': clip_(n.password, 40), 'เลขที่': clip_(n.number, 10), 'ชื่อ': clip_(n.name, 60), 'ห้อง': clip_(n.room, 20) }[h] || '';
        });
        st.sheet.getRange(st.sheet.getLastRow() + 1, 1, 1, row.length).setNumberFormat('@').setValues([row]);
        existing[normUser_(user)] = { username: user, row: st.sheet.getLastRow() };
        added++;
      }
    });
    return json_({ ok: true, added: added, updated: updated });
  } finally {
    lock.releaseLock();
  }
}

function deleteStudent_(data) {
  if (data.key !== TEACHER_KEY) return json_({ ok: false, error: 'unauthorized' });
  var lock = LockService.getScriptLock();
  lock.waitLock(10000);
  try {
    var s = findStudent_(data.username);
    if (!s) return json_({ ok: false, error: 'missing' });
    getStudentSheet_().sheet.deleteRow(s.row);
    return json_({ ok: true });
  } finally {
    lock.releaseLock();
  }
}

// data: { title, status: 'ร่าง' | 'เผยแพร่' } — sets the status on every row of that set
function setExamStatus_(data) {
  if (data.key !== TEACHER_KEY) return json_({ ok: false, error: 'unauthorized' });
  var lock = LockService.getScriptLock();
  lock.waitLock(10000);
  try {
    var sheet = getExamSheet_();
    var header = examHeaders_(sheet, ['สถานะ']);
    var titleCol = header.indexOf('ชุดข้อสอบ');
    var statusCol = header.indexOf('สถานะ');
    var questionCol = header.indexOf('คำถาม');
    var values = readExams_();
    var status = data.status === 'ร่าง' ? 'ร่าง' : 'เผยแพร่';
    var lastTitle = '';
    var count = 0;
    for (var r = 1; r < values.length; r++) {
      lastTitle = values[r][titleCol] || lastTitle;
      if (lastTitle === data.title && values[r][questionCol]) {
        sheet.getRange(r + 1, statusCol + 1).setNumberFormat('@').setValue(status);
        count++;
      }
    }
    return json_({ ok: true, rows: count });
  } finally {
    lock.releaseLock();
  }
}

function doPost(e) {
  var data = JSON.parse(e.postData.contents);
  if (data.action === 'importExams') return importExams_(data);
  if (data.action === 'updateQuestion') return updateQuestion_(data);
  if (data.action === 'deleteQuestion') return deleteQuestion_(data);
  if (data.action === 'uploadImage') return uploadImage_(data);
  if (data.action === 'setExamStatus') return setExamStatus_(data);
  if (data.action === 'studentLogin') return studentLogin_(data);
  if (data.action === 'studentProgress') return studentProgress_(data);
  if (data.action === 'saveProgress') return saveProgress_(data);
  if (data.action === 'listStudents') return listStudents_(data);
  if (data.action === 'saveStudents') return saveStudents_(data);
  if (data.action === 'deleteStudent') return deleteStudent_(data);
  if (!data.student || !data.examId) return json_({ ok: false, error: 'missing fields' });
  var lock = LockService.getScriptLock();
  lock.waitLock(10000);
  try {
    var sheet = getSheet_();
    var clientId = clip_(data.clientId, 40);
    var last = sheet.getLastRow();
    if (clientId && last > 1) {
      var ids = sheet.getRange(2, HEADERS.length, last - 1, 1).getValues();
      for (var i = 0; i < ids.length; i++) {
        if (ids[i][0] === clientId) return json_({ ok: true, duplicate: true });
      }
    }
    sheet.appendRow([
      new Date(data.timestamp || Date.now()),
      clip_(data.student, 40),
      clip_(data.examId, 60),
      clip_(data.examTitle, 120),
      num_(data.score),
      num_(data.answered),
      num_(data.questionCount),
      data.completed === true,
      clip_(data.wrongIds, 500),
      num_(data.durationSec),
      clientId
    ]);
  } finally {
    lock.releaseLock();
  }
  return json_({ ok: true });
}

function doGet(e) {
  if (e.parameter.action === 'exams') {
    return json_({ ok: true, version: 6, values: readExams_() });
  }
  if (!e.parameter.key || e.parameter.key !== TEACHER_KEY) {
    return json_({ ok: false, error: 'unauthorized' });
  }
  var values = getSheet_().getDataRange().getValues();
  var rows = [];
  for (var r = 1; r < values.length; r++) {
    var row = {};
    for (var c = 0; c < HEADERS.length; c++) row[HEADERS[c]] = values[r][c];
    rows.push(row);
  }
  return json_({ ok: true, rows: rows });
}
